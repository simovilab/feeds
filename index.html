GTFS feeds

Feeds live at /<agency>/<feed-type>/<file>. For example: /incofer/schedule/gtfs.zip

.

Each file is overwritten in place, so the URLs below are stable.

{{- $hasFeeds := false }} {{- range $agency := (listFiles "/") }} {{- $agencyPath := printf "/%s" $agency }} {{- if (fileStat $agencyPath).IsDir }} {{- /* Collect first so an agency directory with no feed files renders nothing. */}} {{- $entries := list }} {{- range $feedType := (listFiles $agencyPath) }} {{- $feedTypePath := printf "%s/%s" $agencyPath $feedType }} {{- if (fileStat $feedTypePath).IsDir }} {{- range $file := (listFiles $feedTypePath) }} {{- $path := printf "%s/%s" $feedTypePath $file }} {{- $info := fileStat $path }} {{- if not $info.IsDir }} {{- $href := printf "/%s/%s/%s" (pathEscape $agency) (pathEscape $feedType) (pathEscape $file) }} {{- $entries = append $entries (dict "href" $href "label" (printf "%s/%s/%s" $agency $feedType $file) "info" $info) }} {{- end }} {{- end }} {{- end }} {{- end }} {{- if $entries }} {{- $hasFeeds = true }}

{{ $agency }}

    {{- range $entry := $entries }}
  • {{ $entry.label }} — {{ humanize "size" (printf "%d" $entry.info.Size) }}, updated {{ $entry.info.ModTime.UTC.Format "2006-01-02 15:04 UTC" }}
    • {{- end }}
{{- end }} {{- end }} {{- end }} {{- if not $hasFeeds }}

No feeds are available yet.

{{- end }}

The feeds are compliant with the GTFS Schedule reference and are provided by the Intelligent Mobility Systems Lab (SIMOVI) from the University of Costa Rica.
